cmake_minimum_required(VERSION 3.16)
project(web-wiki VERSION 0.1.2 LANGUAGES NONE)
set(PACKAGE_DIR "lib/web-wiki${PROJECT_VERSION}")
install(FILES pkgIndex.tcl readme.md LICENSE DESTINATION "${PACKAGE_DIR}")
install(DIRECTORY library assets frontend DESTINATION "${PACKAGE_DIR}")
install(FILES MOUNTING.md DESTINATION "${PACKAGE_DIR}")
include(CTest)
if(BUILD_TESTING)
    find_program(TCLSH NAMES tclsh9.1 tclsh9.0 tclsh8.6 tclsh REQUIRED)
    add_test(NAME web-wiki COMMAND "${TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
endif()
